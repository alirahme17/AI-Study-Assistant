from fastapi import FastAPI
from pydantic import BaseModel
from langchain_community.vectorstores import Chroma
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_community.llms import Ollama

# --- 1. Initialize Models and Database ---

# Initialize the embedding model (same as in build_database.py)
print("Loading embedding model...")
model_name = "sentence-transformers/all-MiniLM-L6-v2"
embeddings_model = HuggingFaceEmbeddings(model_name=model_name)

# Define the directory where the vector database is stored
PERSIST_DIRECTORY = "chroma_db"

# Load the existing ChromaDB database
print("Loading ChromaDB...")
db = Chroma(persist_directory=PERSIST_DIRECTORY, embedding_function=embeddings_model)

# Initialize the local LLM with Ollama
# Make sure the model name matches the one you pulled with 'ollama pull'
print("Loading local LLM...")
llm = Ollama(model="llama3.1:8b-instruct-q4_K_M")

# Initialize the FastAPI app
app = FastAPI()

# --- 2. Define API Request Body ---

# Pydantic model to define the expected structure of a request
class Query(BaseModel):
    question: str

# --- 3. Create the API Endpoint ---

@app.post("/api/query")
def query_database(query: Query):
    """
    Receives a question, finds relevant context in the database,
    and generates an answer using an LLM.
    """
    question = query.question
    print(f"Received query: {question}")

    # 4. Search the database for relevant documents
    # The 'k' parameter specifies the number of relevant chunks to retrieve
    retrieved_docs = db.similarity_search(query=question, k=4)
    context = "\n\n".join([doc.page_content for doc in retrieved_docs])
    print("Retrieved context from database.")

    # 5. Create a prompt for the LLM
    # We provide the retrieved context and instruct the LLM to answer based *only* on it.
    prompt_template = f"""
    Answer the following question based only on the context provided.
    If the answer is not in the context, say you don't know.

    Context:
    {context}

    Question:
    {question}
    """
    print("Generating answer...")

    # 6. Get the answer from the LLM
    response = llm.invoke(prompt_template)

    # 7. Return the answer
    return {"answer": response, "sources": [doc.metadata.get("source_file", "unknown") for doc in retrieved_docs]}
